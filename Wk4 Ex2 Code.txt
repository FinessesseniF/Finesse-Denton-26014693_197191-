
function setup() {
  createCanvas(320, 320);
}

function draw() {
  background(179, 36, 0);

///GRID DRAWING - offset so it is just the 'carpet' pattern
  stroke(223, 128, 255)
  strokeWeight(2)
  noFill()
  for(let row = -10; row <= width; row++)
    {for(let col = -10; col <= height; col++)
///use row and col to control size by *ing a number
///change number to get a different size
      {let x = row*40;
       let y = col*40;
         ellipse(x,y,width/2,height/2);
      }
    }
  ///Creates a 'diamond' shape, which I will make into a pattern
  stroke(255, 255, 255)
  fill(255, 204, 0)
// Start drawing the shape.
  beginShape();
  // Add the first anchor point.
        vertex(240, 160);
        // Add the Bézier vertices.
        bezierVertex(160, 160, 160, 160, 160, 240);
        bezierVertex(160, 160, 160, 160, 80, 160);
        bezierVertex(160, 160, 160, 160, 160, 80);
        bezierVertex(160, 160, 160, 160, 240, 160);
  // Stop drawing the shape.
  endShape();
  ///4 Lozenges pointing towards the centre star
    ///topleft
  beginShape();
  vertex(40,40);
      bezierVertex(60, 80, 80, 100, 120, 120);
      bezierVertex(100, 80, 80, 60, 40, 40);
  endShape()
    ///topright
    beginShape();
  vertex(280,40);
      bezierVertex(260, 80, 240, 100, 200, 120);
      bezierVertex(220, 80, 240, 60, 280, 40);
  endShape()    
  ///botleft
  beginShape();
  vertex(40,280);
      bezierVertex(60, 240, 80, 220, 120, 200);
      bezierVertex(100, 240, 80, 260, 40, 280);
  endShape()
    ///botright
    beginShape();
  vertex(280,280);
      bezierVertex(260, 240, 240, 220, 200, 200);
      bezierVertex(220, 240, 240, 260, 280, 280);
  endShape()
  ///This stops the crazy lag
 noLoop()
  ///Screengrabber
function keyPressed(s){
  saveCanvas('grid', 'png');
}
}
